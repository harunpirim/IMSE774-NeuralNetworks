% Under XeLaTeX/LuaLaTeX, Quarto's template loads unicode-math, which turns
% \phi into a math-alphabet character that amsmath's \boldsymbol cannot bold
% ("Improper alphabetic constant"). \symbf is the unicode-math equivalent.
% Aliasing keeps a single \boldsymbol{...} spelling working in the .qmd source
% for both HTML/MathJax and every PDF engine.
\makeatletter
\@ifpackageloaded{unicode-math}{%
  \let\boldsymbol\symbf
}{%
  \usepackage{bm}%
}
\makeatother
